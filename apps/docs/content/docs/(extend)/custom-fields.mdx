Custom field types are **server validators**, not React components. You define how the answer is checked and typed. You render whatever control you want.

---

## 1. Defining a Custom Field Type

Use `defineFieldType()` from `@dimah-form/core` in a shared module (imported by both `dimahForm` and `createFormClient`). Server-only files may import it from `@dimah-form/server`.

```ts title="lib/field-types/rating.ts"
import { defineFieldType } from "@dimah-form/core";
import * as z from "zod";

export const ratingFieldType = defineFieldType({
  type: "rating",

  // 1. Zod schema for validating the field's configuration in defineForm()
  fieldSchema: z.looseObject({
    type: z.literal("rating"),
    min: z.number().int().min(1).default(1),
    max: z.number().int().max(10).default(5),
  }),

  // 2. Server validator for respondent answer values
  validate: (value, field) => {
    if (typeof value !== "number" || !Number.isInteger(value)) {
      return {
        message: "Rating must be an integer",
        code: "EXPECTED_INTEGER",
      };
    }

    const min = field.min ?? 1;
    const max = field.max ?? 5;

    if (value < min) {
      return {
        message: `Rating must be at least ${min}`,
        code: "TOO_SMALL",
        params: { min },
      };
    }

    if (value > max) {
      return {
        message: `Rating cannot exceed ${max}`,
        code: "TOO_LARGE",
        params: { max },
      };
    }

    return undefined; // Valid!
  },

  // 3. Helper to determine if the field is empty for `required` checks
  isEmpty: (value) => value == null,

  // 4. Type token for TypeScript compile-time answer inference
  $Infer: 0 as number,

  // 5. English display string for `formatAnswer` / review
  format: (value, field) => {
    if (typeof value !== "number") return "";
    const max = field.max ?? 5;
    return `${value}/${max}`;
  },
});
```

Export your custom field types from a central module:

```ts title="lib/field-types/index.ts"
import { ratingFieldType } from "./rating";

export const customFieldTypes = [ratingFieldType] as const;
```

## 2. Registering on Server and Client

Register your custom field types in both `dimahForm()` and `createFormClient()`:

### Server Registration

```ts title="lib/form.ts"
import { dimahForm, memoryAdapter } from "@dimah-form/server";
import { customFieldTypes } from "@/lib/field-types";
import { forms } from "@/lib/forms";

export const form = dimahForm({
  database: memoryAdapter(),
  fieldTypes: customFieldTypes,
  forms,
});

export type Form = typeof form;
```

### Client Registration

```ts title="lib/form-client.ts"
"use client";

import { createFormClient } from "@dimah-form/react";
import { customFieldTypes } from "@/lib/field-types";
import type { Form } from "@/lib/form";

export const formClient = createFormClient<Form>({
  basePath: "/api/form",
  fieldTypes: customFieldTypes,
});

export const { useFormResponse } = formClient;
```

A plugin package can ship the same definition on both sides:

```ts
import { definePlugin } from "@dimah-form/server";
import { defineClientPlugin } from "@dimah-form/react";
import { ratingFieldType } from "./rating";

export const ratingPlugin = definePlugin({
  id: "rating",
  fieldTypes: [ratingFieldType],
});

export const ratingClient = defineClientPlugin({
  id: "rating",
  fieldTypes: [ratingFieldType],
});
```

`createFormClient({ plugins: [ratingClient()] })` then installs the local validator without a second `fieldTypes` array. `$Infer` from `createFormClient<Form>()` is still type-only — the client plugin (or `options.fieldTypes`) is required for session checks.

## 3. Using Custom Fields in Forms

`defineForm` stays a loose JSON document at runtime. Bind the catalog to your field types so `min` / `max` (and other `fieldSchema` keys) autocomplete:

```ts title="lib/forms/review.ts"
import { createDefineForm } from "@dimah-form/server";
import type { FormDefinitionUi } from "@dimah-form/ui/types";
import { customFieldTypes } from "@/lib/field-types";

const defineAppForm = createDefineForm({ fieldTypes: customFieldTypes });

export const reviewForm = defineAppForm({
  title: "Product Review",
  slug: "review",
  fields: [
    {
      id: "score",
      type: "rating",
      label: "Star Rating (1-5)",
      required: true,
      min: 1,
      max: 5,
    },
    {
      id: "comments",
      type: "text",
      label: "Review Comments",
    },
  ],
} satisfies FormDefinitionUi<typeof customFieldTypes>);
```

`createDefineForm({ fieldTypes, plugins })` is type-only — still pass the same array and plugins to `dimahForm()` and `createFormClient()`. It types the field document (`min` / `max` here) and plugin `$Meta`. `satisfies FormDefinitionUi<typeof customFieldTypes, typeof plugins>` is for `@dimah-form/ui` `meta` keys plus plugin namespaces; drop it if you do not use that package. TypeScript infers `Form["$Infer"]["answers"]["review"]["score"]` as `number`.

## 4. Rendering Custom Widgets in React

If you use `@dimah-form/ui`, register the widget once — same `type` string as the validator:

```tsx
const widgets = { rating: StarRatingField };

<FormUiProvider widgets={widgets}>
  <FormView form={form} />
</FormUiProvider>;
```

The widget is `{ binding, className, mode? }`. Wrap the control in `FormFieldFrame` (`layout="stack"` | `"choice"` | `"group"`) so `FormUiProvider` `components.FieldFrame` applies. The required mark comes from `components.RequiredMark` when set:

```tsx
import {
  FormFieldFrame,
  fieldNumber,
  type FieldWidgetProps,
} from "@dimah-form/ui";

export function StarRatingField({
  binding,
  className,
  mode,
}: FieldWidgetProps<number>) {
  const field = binding.field;
  if (!field) return null;
  const review = mode === "review";

  return (
    <FormFieldFrame binding={binding} className={className}>
      <StarRating
        id={field.id}
        value={binding.value}
        max={fieldNumber(field, "max") ?? 5}
        disabled={binding.disabled || review}
        onChange={(value) => {
          if (review) return;
          binding.onChange(value);
        }}
      />
    </FormFieldFrame>
  );
}
```

On the headless path, add a branch to your dispatcher:

```tsx title="components/form-field-control.tsx"
import { type FormFieldBinding } from "@dimah-form/react";

export function FormFieldControl(binding: FormFieldBinding) {
  const { field, value, error, disabled, required, onChange } = binding;
  if (!field) return null;

  // Custom Rating Widget
  if (field.type === "rating") {
    const currentRating = typeof value === "number" ? value : 0;
    const max = typeof field.max === "number" ? field.max : 5;

    return (
      <div className="space-y-1.5">
        <label className="block text-sm font-semibold">
          {field.label}
          {required && <span className="ml-1 text-red-500">*</span>}
        </label>
        <div className="flex gap-1">
          {Array.from({ length: max }, (_, i) => i + 1).map((star) => (
            <button
              key={star}
              type="button"
              disabled={disabled}
              onClick={() => onChange(star)}
              className={`text-2xl transition-colors ${
                star <= currentRating ? "text-yellow-400" : "text-gray-300"
              }`}
            >
              ★
            </button>
          ))}
        </div>
        {error && <p className="text-xs text-red-500">{error}</p>}
      </div>
    );
  }

  // Other field types...
  return null;
}
```
